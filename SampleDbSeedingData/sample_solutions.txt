    public static int FindMax(int[] arr) {
        if (arr == null || arr.Length == 0) {
            throw new ArgumentException("Array cannot be empty");

        int max = arr[0];
        for (int i = 1; i < arr.Length; i++) {

    }

----Task-71246 - Подсчет гласных----

// C#
public class Solution {
    public static int CountVowels(string text) {
        if (string.IsNullOrEmpty(text)) return 0;
        int count = 0;
        string vowels = "aeiouAEIOU";
        foreach (char c in text) {
            if (vowels.Contains(c)) count++;
        }
        return count;
    }
}

// Java
public class Solution {
    public static int countVowels(String text) {
        if (text == null) return 0;
        int count = 0;
        String vowels = "aeiouAEIOU";
        for (int i = 0; i < text.length(); i++) {
            if (vowels.indexOf(text.charAt(i)) != -1) count++;
        }
        return count;
    }
}

// Kotlin
class Solution {
    companion object {
        fun countVowels(text: String?): Int {
            if (text.isNullOrEmpty()) return 0
            val vowels = setOf('a', 'e', 'i', 'o', 'u', 'A', 'E', 'I', 'O', 'U')
            return text.count { it in vowels }
        }
    }
}

// TypeScript
export class Solution {
    static countVowels(text: string | null): number {
        if (!text) return 0;
        const vowels = /[aeiou]/gi;
        const matches = text.match(vowels);
        return matches ? matches.length : 0;
    }
}

// Node.js

        if (!text) return 0;

    }

    public static int[] SortArray(int[] arr) {
        if (arr == null) return new int[0];
        int[] sorted = new int[arr.Length];

// C#
public class Solution {
    public static bool Solution(string str) {
        if (str == null) return false;
        string cleaned = new string(str.ToLower().Where(char.IsLetterOrDigit).ToArray());
        return cleaned.SequenceEqual(cleaned.Reverse());
    }
}

// Java
public class Solution {
    public static boolean Solution(String str) {
        if (str == null) return false;
        String cleaned = str.toLowerCase().replaceAll("[^a-z0-9]", "");
        return new StringBuilder(cleaned).reverse().toString().equals(cleaned);
    }
}

        }
    }
}

// C#
public class Solution {
    public static string[] Solution(string[] items) {
        if (items == null) return new string[0];
        var seen = new HashSet<string>();
        var result = new List<string>();
        foreach (var item in items) {
            if (!seen.Contains(item)) {
                seen.Add(item);
                result.Add(item);
            }
        }
        return result.ToArray();
    }
}

// C#
public class Solution {
    public static int[] Solution(int[] arr1, int[] arr2) {
        List<int> list = new List<int>();
        if (arr1 != null) list.AddRange(arr1);
        if (arr2 != null) list.AddRange(arr2);
        return list.ToArray();
    }
}

// Java
import java.util.ArrayList;
import java.util.List;

public class Solution {
    public static int[] Solution(int[] arr1, int[] arr2) {
        List<Integer> list = new ArrayList<>();
        if (arr1 != null) {
            for (int num : arr1) list.add(num);
        }
        if (arr2 != null) {
            for (int num : arr2) list.add(num);
        }
        return list.stream().mapToInt(i -> i).toArray();
    }
}

// Kotlin
class Solution {
    companion object {
        fun Solution(arr1: IntArray?, arr2: IntArray?): IntArray {
            val list = mutableListOf<Int>()
            arr1?.forEach { list.add(it) }
            arr2?.forEach { list.add(it) }
            return list.toIntArray()
        }
    }
}
